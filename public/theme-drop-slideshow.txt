<script>
  var slideSpeed = {{ section.settings.slide_speed }} * 1000;
  var slideDuration = {{ section.settings.slide_duration }} * 1000;    
  var autoplay = {{ section.settings.slide_autoplay }};

  function initializeSlideshow () {
    $('.slideshow-carousel-{{ section.id }}').owlCarousel({
      nav: true,
      loop: {{ section.settings.slide_loop }},
      autoplay: {{ section.settings.slide_autoplay }},
      autoplayTimeout: slideDuration,
      autoplaySpeed: slideSpeed,
      navText: [
        '<i class=\'material-icons\'></i>',
        '<i class=\'material-icons\'></i>'
      ],
      items: 1,
      itemsDesktop: false,
      itemsDesktopSmall: false,
      itemsTablet: false,
      itemsMobile: false,
      onInitialized: function() {
        updateTabIndexes();
      },
      onTranslated: function() {
        updateTabIndexes();
      }
    });

    $(document).on('keydown', function(event) {
      if (event.key === 'ArrowLeft') {
        $('.slideshow-carousel-{{ section.id }}').trigger('prev.owl.carousel');
      } else if (event.key === 'ArrowRight') {
        $('.slideshow-carousel-{{ section.id }}').trigger('next.owl.carousel');
      }
    });
    
    if (autoplay) {
      $('#toggle-autoplay').text('Stop Autoplay');
    } else {
      $('#toggle-autoplay').text('Start Autoplay');
    }
    $('#toggle-autoplay').click(function() {
      if (autoplay) {
        $('.slideshow-carousel-{{ section.id }}').trigger('stop.owl.autoplay');
        autoplay = false;
        $(this).text('Start Autoplay');
      } else {
        $('.slideshow-carousel-{{ section.id }}').trigger('play.owl.autoplay', [slideDuration]);
        autoplay = true;
        $(this).text('Stop Autoplay');
      }
    });
  }

  jQuery(document).ready(initializeSlideshow);  
  document.addEventListener("shopify:section:load", initializeSlideshow);

</script>
